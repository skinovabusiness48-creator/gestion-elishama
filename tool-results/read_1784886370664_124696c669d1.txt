   197→  cart: CartItem[];
   198→  note: string;
   199→  submitting: boolean;

   201→  onInc: (key: string) => void;

   203→  onRemove: (key: string) => void;

   209→
   210→function CartPanel({
   211→  cart,
   212→  note,
   213→  submitting,
   214→  onNoteChange,
   215→  onInc,
   216→  onDec,
   217→  onRemove,
   218→  onPriceChange,
   219→  onClear,
   220→  onSubmit,
   221→  variant = "inline",
   222→}: CartPanelProps) {
   223→  const total = cart.reduce((s, c) => s + lineTotal(c), 0);
   224→  const totalQty = cart.reduce((s, c) => s + c.quantity, 0);
   225→
   226→  return (
   227→    <Card className="flex h-full flex-col">
   228→      <CardHeader className="pb-3">
   229→        <div className="flex items-center justify-between gap-2">
   230→          <CardTitle className="flex items-center gap-2 text-base">
   231→            <ShoppingCart className="h-4 w-4 text-primary" />
   232→            Panier
   233→            {cart.length > 0 && (
   234→              <Badge variant="secondary" className="font-mono tabular-nums">
   235→                {formatNumber(totalQty)}
   236→              </Badge>
   237→            )}
   238→          </CardTitle>
   239→          {cart.length > 0 && (
   240→            <Button
   241→              variant="ghost"
   242→              size="sm"
   243→              className="h-7 px-2 text-xs text-muted-foreground hover:text-destructive"

   246→              <Trash2 className="h-3.5 w-3.5" />
   247→              Vider
   248→            </Button>
   249→          )}
   250→        </div>
   251→      </CardHeader>
   252→      <CardContent className="flex flex-1 flex-col gap-3 overflow-hidden">
   253→        {cart.length === 0 ? (
   254→          <EmptyState
   255→            title="Panier vide"
   256→            description="Ajoutez des plats, boissons ou produits pour composer une vente."
   257→            className="flex-1 py-8"
   258→            icon={<ShoppingCart className="h-6 w-6" />}
   259→          />

   261→          <ScrollArea
   262→            className={cn(
   263→              "flex-1 pr-2",
   264→              variant === "inline" ? "max-h-[44vh] lg:max-h-[52vh]" : "max-h-[52vh]",

   267→            <ul className="space-y-2">

   269→                const k = cartKey(c);

   271→                  <li key={k} className="rounded-lg border bg-card/60 p-2.5">
   272→                    <div className="flex items-start gap-2">
   273→                      <div className="min-w-0 flex-1">
   274→                        <p className="truncate text-sm font-medium">{c.itemName}</p>
   275→                        <p className="text-xs text-muted-foreground">
   276→                          {formatNumber(c.quantity)} × {formatMoney(c.unitPrice)}
   277→                        </p>
   278→                      </div>
   279→                      <Button
   280→                        variant="ghost"
   281→                        size="icon"
   282→                        className="h-6 w-6 shrink-0 text-muted-foreground hover:text-destructive"
   283→                        onClick={() => onRemove(k)}
   284→                      >
   285→                        <X className="h-3.5 w-3.5" />
   286→                      </Button>
   287→                    </div>
   288→                    <div className="mt-2 flex items-center gap-2">
   289→                      <div className="flex items-center rounded-md border">
   290→                        <Button
   291→                          variant="ghost"
   292→                          size="icon"
   293→                          className="h-7 w-7"
   294→                          onClick={() => onDec(k)}

   303→                          variant="ghost"

   307→                        >
   308→                          <Plus className="h-3.5 w-3.5" />
   309→                        </Button>
   310→                      </div>
   311→                      <Input
   312→                        type="number"
   313→                        min={0}
   314→                        step="any"
   315→                        value={c.unitPrice}
   316→                        onChange={(e) => onPriceChange(k, Number(e.target.value) || 0)}
   317→                        className="ml-auto h-8 w-28 text-right text-sm tabular-nums"
   318→                        aria-label="Prix unitaire"
   319→                      />

   321→                        {formatMoney(lineTotal(c))}
   322→                      </div>
   323→                    </div>
   324→                  </li>
   325→                );
   326→              })}
   327→            </ul>
   328→          </ScrollArea>
   329→        )}
   330→
   331→        <div className="space-y-3 border-t pt-3">
   332→          <div>

   349→            </span>
   350→          </div>

   352→            className="w-full"
   353→            size="lg"
   354→            onClick={onSubmit}
   355→            disabled={cart.length === 0 || submitting}

   357→            {submitting ? (
   358→              <Loader2 className="h-4 w-4 animate-spin" />

   362→            Enregistrer la vente
   363→          </Button>
   364→        </div>
   365→      </CardContent>
   366→    </Card>
   367→  );

   393→
   394→interface POSPanelProps {
   395→  onSaleSaved: () => void;
   396→}
   397→
   398→function POSPanel({ onSaleSaved }: POSPanelProps) {
   399→  const [products, setProducts] = useState<Product[]>([]);
   400→  const [dishes, setDishes] = useState<Dish[]>([]);
   401→  const [drinks, setDrinks] = useState<Drink[]>([]);
   402→  const [loading, setLoading] = useState(true);
   403→  const [posTab, setPosTab] = useState<string>("dishes");
   404→  const [search, setSearch] = useState("");
   405→  const [cart, setCart] = useState<CartItem[]>([]);
   406→  const [note, setNote] = useState("");
   407→  const [submitting, setSubmitting] = useState(false);
   408→  const [mobileCartOpen, setMobileCartOpen] = useState(false);
   409→
   410→  const loadCatalog = useCallback(async () => {
   411→    try {
   412→      setLoading(true);
   413→      const [p, d, dr] = await Promise.all([
   414→        apiFetch<Product[]>("/api/products?available=true"),
   415→        apiFetch<Dish[]>("/api/dishes"),
   416→        apiFetch<Drink[]>("/api/drinks"),
   417→      ]);
   418→      setProducts(p);
   419→      setDishes(d.filter((x) => x.available !== false));
   420→      setDrinks(dr.filter((x) => x.available !== false));
   421→    } catch (err) {
   422→      toast.error(
   423→        err instanceof Error ? err.message : "Impossible de charger le catalogue",
   424→      );
   425→    } finally {
   426→      setLoading(false);
   427→    }
   428→  }, []);
   429→
   430→  useEffect(() => {
   431→    loadCatalog();
   432→  }, [loadCatalog]);
   433→
   434→  const filteredDishes = useMemo(() => {
   435→    const q = search.trim().toLowerCase();
   436→    if (!q) return dishes;
   437→    return dishes.filter((d) => d.name.toLowerCase().includes(q));
   438→  }, [dishes, search]);
   439→
   440→  const filteredDrinks = useMemo(() => {
   441→    const q = search.trim().toLowerCase();
   442→    if (!q) return drinks;
   443→    return drinks.filter((d) => d.name.toLowerCase().includes(q));
   444→  }, [drinks, search]);
   445→
   446→  const filteredProducts = useMemo(() => {
   447→    const q = search.trim().toLowerCase();
   448→    if (!q) return products;
   449→    return products.filter((p) => p.name.toLowerCase().includes(q));
   450→  }, [products, search]);
   451→
   452→  const addToCart = useCallback((item: CartItem) => {
   453→    setCart((prev) => {
   454→      const k = cartKey(item);
   455→      const idx = prev.findIndex((c) => cartKey(c) === k);
   456→      if (idx >= 0) {
   457→        const next = [...prev];
   458→        next[idx] = { ...next[idx], quantity: next[idx].quantity + 1 };
   459→        return next;
   460→      }
   461→      return [...prev, item];
   462→    });
   463→  }, []);
   464→
   465→  const inc = (key: string) =>
   466→    setCart((prev) =>
   467→      prev.map((c) => (cartKey(c) === key ? { ...c, quantity: c.quantity + 1 } : c)),
   468→    );
   469→  const dec = (key: string) =>
   470→    setCart((prev) =>
   471→      prev.map((c) =>
   472→        cartKey(c) === key ? { ...c, quantity: Math.max(1, c.quantity - 1) } : c,
   473→      ),
   474→    );
   475→  const removeItem = (key: string) =>
   476→    setCart((prev) => prev.filter((c) => cartKey(c) !== key));
   477→  const priceChange = (key: string, v: number) =>
   478→    setCart((prev) =>
   479→      prev.map((c) => (cartKey(c) === key ? { ...c, unitPrice: Math.max(0, v) } : c)),
   480→    );
   481→  const clear = () => {
   482→    setCart([]);
   483→    setNote("");
   484→  };
   485→
   486→  const submit = async () => {
   487→    if (cart.length === 0) return;
   488→    setSubmitting(true);
   489→    try {
   490→      const items = cart.map((c) => {
   491→        const base: Record<string, unknown> = {
   492→          itemType: c.itemType,
   493→          itemName: c.itemName,
   494→          quantity: c.quantity,
   495→          unitPrice: c.unitPrice,
   496→          total: lineTotal(c),
   497→        };
   498→        if (c.itemType === "product") base.productId = c.refId;
   499→        else if (c.itemType === "dish") base.dishId = c.refId;
   500→        else base.drinkId = c.refId;
   501→        return base;
   502→      });
   503→      await apiFetch("/api/sales", {
   504→        method: "POST",
   505→        body: JSON.stringify({ note: note.trim() || undefined, items }),
   506→      });
   507→      toast.success("Vente enregistrée avec succès");
   508→      clear();
   509→      setMobileCartOpen(false);
   510→      onSaleSaved();
   511→    } catch (err) {
   512→      toast.error(
   513→        err instanceof Error ? err.message : "Erreur lors de l'enregistrement",
   514→      );
   515→    } finally {
   516→      setSubmitting(false);
   517→    }
   518→  };
   519→
   520→  const cartProps: CartPanelProps = {
   521→    cart,
   522→    note,
   523→    submitting,
   524→    onNoteChange: setNote,
   525→    onInc: inc,
   526→    onDec: dec,
   527→    onRemove: removeItem,
   528→    onPriceChange: priceChange,
   529→    onClear: clear,
   530→    onSubmit: submit,
   531→  };
   532→
   533→  const totalQty = cart.reduce((s, c) => s + c.quantity, 0);
   534→
   535→  return (
   536→    <div className="grid grid-cols-1 gap-4 lg:grid-cols-3">
   537→      {/* Catalogue (colonne gauche) */}
   538→      <div className="lg:col-span-2">
   539→        <Card>
   540→          <CardHeader className="pb-3">
   541→            <div className="flex flex-col gap-3 sm:flex-row sm:items-center sm:justify-between">
   542→              <CardTitle className="text-base">Catalogue</CardTitle>
   543→              <div className="relative w-full sm:max-w-xs">
   544→                <Search className="absolute left-2.5 top-1/2 h-4 w-4 -translate-y-1/2 text-muted-foreground" />
   545→                <Input
   546→                  value={search}
   547→                  onChange={(e) => setSearch(e.target.value)}
   548→                  placeholder="Rechercher un article…"
   549→                  className="pl-9"
   550→                />
   551→              </div>
   552→            </div>
   553→          </CardHeader>
   554→          <CardContent>
   555→            <Tabs value={posTab} onValueChange={setPosTab}>
   556→              <TabsList className="grid w-full grid-cols-3 sm:inline-flex sm:w-auto">
   557→                <TabsTrigger value="dishes">
   558→                  <UtensilsCrossed className="h-4 w-4" />
   559→                  <span className="hidden sm:inline">Plats</span>
   560→                  <Badge variant="secondary" className="ml-1 font-mono">
   561→                    {dishes.length}
   562→                  </Badge>
   563→                </TabsTrigger>
   564→                <TabsTrigger value="drinks">
   565→                  <Wine className="h-4 w-4" />
   566→                  <span className="hidden sm:inline">Boissons</span>
   567→                  <Badge variant="secondary" className="ml-1 font-mono">
   568→                    {drinks.length}
   569→                  </Badge>
   570→                </TabsTrigger>
   571→                <TabsTrigger value="products">
   572→                  <Package className="h-4 w-4" />
   573→                  <span className="hidden sm:inline">Produits</span>
   574→                  <Badge variant="secondary" className="ml-1 font-mono">
   575→                    {products.length}
   576→                  </Badge>
   577→                </TabsTrigger>
   578→              </TabsList>
   579→
   580→              <TabsContent value="dishes" className="mt-4">
   581→                {loading ? (
   582→                  <SkeletonGrid />
   583→                ) : filteredDishes.length === 0 ? (
   584→                  <EmptyState
   585→                    title="Aucun plat disponible"
   586→                    description="Ajoutez des plats dans le module Plats ou ajustez votre recherche."
   587→                    icon={<UtensilsCrossed className="h-6 w-6" />}
   588→                  />
   589→                ) : (
   590→                  <div className="grid grid-cols-2 gap-2.5 sm:grid-cols-3 xl:grid-cols-4">
   591→                    {filteredDishes.map((d) => (
   592→                      <ItemButton
   593→                        key={d.id}
   594→                        name={d.name}
   595→                        price={d.price}
   596→                        photo={d.photo}
   597→                        icon={<UtensilsCrossed className="h-8 w-8" />}
   598→                        onClick={() =>
   599→                          addToCart({
   600→                            itemType: "dish",
   601→                            refId: d.id,
